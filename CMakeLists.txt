cmake_minimum_required(VERSION 3.13)
project(mx CXX)

# Clean-slate spike (newgen). The generated mx/core library and most of the C++
# were removed so the code generator can be rebuilt (see AGENTS.md). What
# survives is the structural skeleton: the embedded ezxml XML layer and the core
# roundtrip (corert) test harness. corert does not compile until the new
# generator emits src/private/mx/core -- expected on this branch.

set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_EXPORT_COMPILE_COMMANDS ON)
set(EXECUTABLE_OUTPUT_PATH ${CMAKE_BINARY_DIR})
set(LIBRARY_OUTPUT_PATH ${CMAKE_BINARY_DIR})

option(MX_CORE_DEV "Build the core roundtrip (corert) test binary against the regenerated mx/core." OFF)

set(PRIVATE_DIR "${CMAKE_CURRENT_SOURCE_DIR}/src/private")

# ezxml: the embedded XML reader/writer (pugixml-backed, self-contained). The
# structural XML layer the regenerated core builds on. Builds today.
file(GLOB_RECURSE SRC_MX_EZXML ${PRIVATE_DIR}/mx/ezxml/*.cpp ${PRIVATE_DIR}/mx/ezxml/*.h*)
add_library(mx-ezxml STATIC ${SRC_MX_EZXML})
target_include_directories(mx-ezxml PUBLIC
    ${PRIVATE_DIR}
    ${PRIVATE_DIR}/mx/ezxml/src/include
    ${PRIVATE_DIR}/mx/ezxml/src/private)

if(MX_CORE_DEV)

    # The generator's output. Empty now; the globs fill in as codegen lands
    # mx/core (and any mx/utility it needs). Compiled straight into the test
    # binary so an empty set is not a CMake error but a compile error in corert
    # (missing mx/core/Document.h) -- the intended "fails now" state.
    file(GLOB_RECURSE SRC_MX_CORE    ${PRIVATE_DIR}/mx/core/*.c* ${PRIVATE_DIR}/mx/core/*.h*)
    file(GLOB_RECURSE SRC_MX_UTILITY ${PRIVATE_DIR}/mx/utility/*.cpp ${PRIVATE_DIR}/mx/utility/*.h)

    # corert: round-trips each data/ file through mx::core and compares against a
    # normalized, fixup-adjusted form of the input. Fixer applies .fixup.xml
    # sidecars; cpul is the Catch test runner (main); the import/ helpers
    # normalize attribute order and decimals and depend only on ezxml.
    file(GLOB_RECURSE SRC_CORERT ${PRIVATE_DIR}/mxtest/corert/*.cpp ${PRIVATE_DIR}/mxtest/corert/*.h)
    file(GLOB_RECURSE SRC_CPUL   ${PRIVATE_DIR}/cpul/*.cpp ${PRIVATE_DIR}/cpul/*.h)
    set(SRC_CORERT_HELPERS
        ${PRIVATE_DIR}/mxtest/import/ChangeValues.cpp
        ${PRIVATE_DIR}/mxtest/import/ChangeValues.h
        ${PRIVATE_DIR}/mxtest/import/SortAttributes.cpp
        ${PRIVATE_DIR}/mxtest/import/SortAttributes.h
        ${PRIVATE_DIR}/mxtest/import/DecimalFields.h)

    # PathRoot.h hands the corert file walker the repo root at configure time.
    file(WRITE ${PRIVATE_DIR}/mxtest/file/PathRoot.h
"// Auto-generated by CMake.
#pragma once
#ifndef MX_REPO_ROOT_PATH
#define MX_REPO_ROOT_PATH \"${CMAKE_CURRENT_SOURCE_DIR}\"
#endif
")

    find_package(Threads)
    add_executable(mxtest-core-dev
        ${SRC_CORERT} ${SRC_CORERT_HELPERS} ${SRC_CPUL}
        ${SRC_MX_CORE} ${SRC_MX_UTILITY})
    target_include_directories(mxtest-core-dev PRIVATE
        ${PRIVATE_DIR}
        ${PRIVATE_DIR}/mx/ezxml/src/include
        ${PRIVATE_DIR}/mx/ezxml/src/private)
    target_link_libraries(mxtest-core-dev mx-ezxml ${CMAKE_THREAD_LIBS_INIT})

endif()
